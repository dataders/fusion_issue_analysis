Bakeoff fit

  • Canonical face: transform/faces/fusion-issue-health.yml; this directory mirrors it.
  • Queries dbt dashboard models directly via SQL — summary_kpis, issue_triage_health, velocity, age_distribution, community_priorities, and more.
  • New upstream columns wired: net_flow_4w, bucket_sort_order, issue_url for links.
  • Chart types: kpi, area, line, stacked bar, and table with link click-throughs.
  • Age distribution derives bucket order from data (ORDER BY bucket_sort_order) — no hardcoded label list.
  • Local preview: cd transform && dft serve faces/fusion-issue-health.yml
  • Validate: cd transform && dft validate faces/fusion-issue-health.yml — passes.

title: "dbt-fusion Issue Health"
source: fusion_issues

queries:
  triage_health_kpis:
    sql: >
      SELECT slipped_through_count, triage_queue_count,
             hard_blocker_count, hard_blocker_unreleased,
             stale_count AS triage_stale_count,
             needs_repro_count, repro_verified_count, total_open
      FROM fusion_issues.main.issue_triage_health

  summary_kpis:
    sql: >
      SELECT open_issues, net_flow_4w,
             rolling_median_close_days, pct_responded_48h,
             stale_count AS summary_stale_count
      FROM fusion_issues.main.summary_kpis

  velocity:
    sql: >
      SELECT week, issue_category, median_days
      FROM fusion_issues.main.velocity
      ORDER BY week, issue_category

  age_distribution:
    sql: >
      SELECT age_bucket, bucket_sort_order,
             issue_category, issue_count
      FROM fusion_issues.main.age_distribution
      ORDER BY bucket_sort_order, issue_category

  community_priorities:
    sql: >
      SELECT issue_number, title, issue_category,
             reactions_total_count, age_days, issue_url
      FROM fusion_issues.main.community_priorities
      ORDER BY reactions_total_count DESC

  # ... (oldest_untriaged, cumulative_flow, response_pctiles,
  #      close_by_label, assignee_workload, label_health_kpis)

charts:
  net_flow_kpi:
    query: summary_kpis
    type: kpi
    value: net_flow_4w
    label: "Net Flow (4 wk)"

  velocity_chart:
    query: velocity
    type: line
    x: week
    y: median_days
    color: issue_category
    title: "Median Days to Close: Bugs vs Enhancements"

  age_distribution_chart:
    query: age_distribution
    type: bar
    x: age_bucket
    y: issue_count
    color: issue_category
    stack: true
    sort: { by: bucket_sort_order, order: asc }
    title: "Open Issue Age by Type"

  community_priorities_table:
    query: community_priorities
    type: table
    href: issue_url   # links from issue_url column — never hardcoded
    # ... column labels omitted for brevity

rows:
  - title: "Operational Triage"
    cols: [slipped_through_kpi, triage_queue_kpi, hard_blockers_kpi,
           triage_stale_kpi, needs_repro_kpi, repro_verified_kpi, total_open_kpi]
  - oldest_untriaged_table
  - title: "Key Metrics"
    cols: [open_issues_kpi, net_flow_kpi, median_close_kpi,
           response_sla_kpi, summary_stale_kpi]
  - title: "Trends"
    rows:
      - cols: [cumulative_flow_chart, velocity_chart]
      - cols: [response_pctiles_chart, age_distribution_chart]
      - close_by_label_chart
  - title: "Triage Health"
    cols: [pct_labeled_kpi, pct_typed_kpi, pct_assigned_kpi, pct_milestoned_kpi]
  - title: "People"
    rows: [assignee_workload_chart, community_priorities_table]
